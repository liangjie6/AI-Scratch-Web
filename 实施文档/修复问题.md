```python
@staticmethod
async def _parse_responses_lines(
    lines: AsyncIterator[str],
) -> AsyncIterator[AgnesStreamDelta]:
    async for line in lines:
        line = line.strip()

        if not line or line.startswith(':') or line.startswith('event:'):
            continue

        if not line.startswith('data:'):
            continue

        raw = line[5:].strip()

        if raw == '[DONE]':
            return

        try:
            item = json.loads(raw)
        except json.JSONDecodeError as exc:
            raise AgnesTextProviderError(
                'GPT returned an invalid Responses stream'
            ) from exc

        if not isinstance(item, dict):
            continue

        event_type = item.get('type')

        if event_type in {
            'error',
            'response.failed',
            'response.incomplete',
        } or item.get('error'):
            raise AgnesTextProviderError(
                'GPT Responses stream failed'
            )

        if event_type == 'response.output_text.delta':
            text = item.get('delta')
            if isinstance(text, str) and text:
                yield AgnesStreamDelta('content', text)

        elif event_type in {
            'response.reasoning_summary_text.delta',
            'response.reasoning_text.delta',
        }:
            text = item.get('delta')
            if isinstance(text, str) and text:
                yield AgnesStreamDelta('reasoning', text)

        elif event_type == 'response.completed':
            response = item.get('response')

            usage = (
                response.get('usage')
                if isinstance(response, dict)
                else item.get('usage')
            )

            if isinstance(usage, dict):
                normalized_usage = {
                    key: value
                    for key, value in (
                        ('input_tokens', usage.get('input_tokens')),
                        ('output_tokens', usage.get('output_tokens')),
                        ('total_tokens', usage.get('total_tokens')),
                    )
                    if isinstance(value, int)
                }

                if normalized_usage:
                    yield AgnesStreamDelta(
                        'content',
                        '',
                        normalized_usage,
                    )
```

```python
async def _collect_resilient(
    provider: AgnesTextProvider,
    messages: list[dict[str, str]],
    thinking: ThinkingSettings,
    *,
    retries: int = 2,
    retry_delay_seconds: float = 0.5,
    json_output: bool = False,
) -> tuple[str, str]:
    """Retry buffered provider requests after transient stream failures."""
    current_thinking = thinking
    for attempt in range(retries + 1):
        try:
            return await _collect(provider, messages, current_thinking, json_output=json_output)
        except AgnesTextProviderError as exc:
            if attempt >= retries:
                raise
            delay = retry_delay_seconds * (2 ** attempt)
            logger.warning(
                'ai_provider_retry provider=%s attempt=%s max_attempts=%s delay_seconds=%s error=%s',
                getattr(provider, 'provider', 'unknown'), attempt + 2, retries + 1, delay,
                _error_summary(exc),
            )
            current_thinking = ThinkingSettings(enabled=False, strength='low')
            if delay > 0:
                await asyncio.sleep(delay)
    raise RuntimeError('AI provider collection produced no result')
```

这段代码依赖的 `asyncio`、`logger`、`_error_summary` 当前文件中已经存在，不需要新增 import。

具体行为：

- 第一次请求使用原始推理配置。
- 第一次失败后等待 `0.5` 秒。
- 第二次失败后等待 `1.0` 秒。
- 最多执行 3 次请求。
- 第 2、3 次请求关闭推理，并使用 `strength='low'`。
- `_collect` 内部缓存的是当前请求的局部结果，因此断流产生的半截 JSON 不会和下一次结果拼接。

**1. 新增吸附模块**

文件：[`frontend/src/canvas/model/alignment-snapping.ts`](D:\ywj\AI-Scratch-web\frontend\src\canvas\model\alignment-snapping.ts)

完整内容：

```ts
import type {Matrix2D, Point, Rect} from './canvas-types';

export interface AlignmentGuides {
  x: number[];
  y: number[];
}

export interface TranslationSnapResult {
  transform: Matrix2D;
  guides: AlignmentGuides;
}

type AxisSnap = {delta: number; coordinate: number; distance: number};

function transformPoint(matrix: Matrix2D, point: Point): Point {
  return {
    x: matrix.a * point.x + matrix.c * point.y + matrix.e,
    y: matrix.b * point.x + matrix.d * point.y + matrix.f
  };
}

export function preciseTransformedRectBounds(bounds: Rect, transform: Matrix2D): Rect {
  const corners = [
    {x: bounds.x, y: bounds.y},
    {x: bounds.x + bounds.width, y: bounds.y},
    {x: bounds.x, y: bounds.y + bounds.height},
    {x: bounds.x + bounds.width, y: bounds.y + bounds.height}
  ].map(point => transformPoint(transform, point));

  const left = Math.min(...corners.map(point => point.x));
  const top = Math.min(...corners.map(point => point.y));
  const right = Math.max(...corners.map(point => point.x));
  const bottom = Math.max(...corners.map(point => point.y));

  return {
    x: left,
    y: top,
    width: right - left,
    height: bottom - top
  };
}

function anchors(start: number, size: number): number[] {
  return [start, start + size / 2, start + size];
}

function closestSnap(
  moving: number[],
  targets: number[],
  threshold: number
): AxisSnap | null {
  let closest: AxisSnap | null = null;

  for (const movingCoordinate of moving) {
    for (const targetCoordinate of targets) {
      const delta = targetCoordinate - movingCoordinate;
      const distance = Math.abs(delta);

      if (
        distance <= threshold &&
        (!closest || distance < closest.distance)
      ) {
        closest = {
          delta,
          coordinate: targetCoordinate,
          distance
        };
      }
    }
  }

  return closest;
}

export function snapLayerTranslation(
  bounds: Rect,
  proposedTransform: Matrix2D,
  targetRects: Rect[],
  threshold: number
): TranslationSnapResult {
  const moving = preciseTransformedRectBounds(bounds, proposedTransform);

  const targetX = targetRects.flatMap(rect =>
    anchors(rect.x, rect.width)
  );

  const targetY = targetRects.flatMap(rect =>
    anchors(rect.y, rect.height)
  );

  const xSnap = closestSnap(
    anchors(moving.x, moving.width),
    targetX,
    threshold
  );

  const ySnap = closestSnap(
    anchors(moving.y, moving.height),
    targetY,
    threshold
  );

  return {
    transform: {
      ...proposedTransform,
      e: proposedTransform.e + (xSnap?.delta ?? 0),
      f: proposedTransform.f + (ySnap?.delta ?? 0)
    },
    guides: {
      x: xSnap ? [xSnap.coordinate] : [],
      y: ySnap ? [ySnap.coordinate] : []
    }
  };
}
```

项目模式流程：


每个 target 必须串行处理，不能并发：

1. 根据逻辑 target 找到真实 VM target。
2. 设置 VM editing target。
3. 重新读取最新 `AiTargetSnapshot`。
4. 携带当前 `program_target` 和 manifest 发起请求。
5. 收到单个 ProgramIR 后立即校验和编译。
6. 调用已有 `applyAiProgram` 渐进创建 block。
7. 记录 `appliedPrograms`。
8. 成功后继续下一个 target。

`shared_state` 在程序生成前写入舞台：

- 全局变量创建在舞台变量表。
- 全局列表创建在舞台变量表。
- 广播名称创建为全局广播。
- 已存在的同名定义直接复用。
- 不把真实 VM ID 传回模型。

失败和取消：

- 资源或任一 target 程序失败时恢复最初 backup。
- 当前 ProgramIR 局部注入失败先执行局部回滚。
- 局部回滚失败再执行项目 backup 恢复。

```ts
  if (output instanceof Uint8Array) return new Uint8Array(output);
  // scratch-vm emits a Blob; the declared type is narrower than runtime.
  return new Uint8Array(await (output as Blob).arrayBuffer());
}
```

```python
        async with asyncio.timeout(get_settings().ai_blocks_provider_timeout_seconds):
            stream_kwargs = {
                'thinking_enabled': thinking.enabled,
                'thinking_strength': thinking.strength,
            }
            if hasattr(provider, 'provider'):
                stream_kwargs['json_output'] = json_output

            async for delta in provider.stream_with_reasoning(messages, **stream_kwargs):
                if delta.usage:
                    usage.update(delta.usage)
                    continue

                delta_count += 1

                # 只输出 content，且不自动换行。
                if delta.kind == 'content':
                    print(delta.text, end='', flush=True)

                # reasoning 仍继续向前端传递，但不写入控制台。
                yield delta

        input_tokens = usage.get('input_tokens', 0)
        output_tokens = usage.get('output_tokens', 0)
        total_tokens = usage.get(
            'total_tokens',
            input_tokens + output_tokens,
        )

        print(
            '\n'
            'ai_stream_summary '
            f"provider={getattr(provider, 'provider', 'unknown')} "
            f"input_tokens={input_tokens} "
            f"output_tokens={output_tokens} "
            f"total_tokens={total_tokens} "
            f"duration_ms={round((asyncio.get_running_loop().time() - started_at) * 1000)}",
            flush=True,
        )
```

注意，这里不要使用：

```python
logger.info(...)
```

因为 `logger.info` 会自动带上：

```text
时间戳 日志级别 logger名称
```

这里必须使用：

```python
print(..., flush=True)
```

`ai_stream_started`、请求开始、请求失败等其他诊断日志仍会继续使用标准 logger 格式；本次修改只针对流式内容和流式汇总行。

**重修复时必须优先注意的真实缺口**

1. **当前事务 revision 检查可能误伤多操作计划**

   `project-resource-runner.ts` 在每个资源/程序操作前都用最初的 `snapshot.revision` 检查。但 `scratch-gui-bridge.ts` 的资源操作会触发 `PROJECT_CHANGED` 并增加 revision。

   因此多资源 ProjectPlan 可能在第一项成功后，第二项因为 revision 改变而被当成用户外部修改，直接恢复 backup。需要区分：

   - 事务自身产生的 revision
   - 用户在事务期间产生的 revision

2. **`add_sprite` 的 generated costume 数组还没有真正支持**

   当前 `project-resource-preparer.ts` 对 `add_sprite` 和 `add_builtin_sprite` 共用分支，并要求从 `selections` 中选择 Scratch 角色：

   ```ts

     const selected = options.selections?.get(op.op_id);
     ...
   }
   ```

   这会忽略 `add_sprite.costumes` 中的 `image_generation` 来源。需要分别处理：

   - `add_builtin_sprite`：使用本地 Scratch 角色候选。
   - `add_sprite`：按每个 costume 的 source 分别 Prepare，可混合本地素材和生成素材。

3. **新增角色的 `x/y` 当前没有传入 VM**

   `ResourceOp` 定义了 `x/y/size`，但 executor 只把 `size` 放进 `SpriteExport.initialSize`。底层 `buildSpriteImport` 当前把新角色位置固定为 `x: 0, y: 0`。

   `_timed_events` 当前日志包含消息内容和 delta 文本，而文档要求日志不能包含：

   - 用户消息
   - 程序内容
   - 图片
   - model delta
   - 内部 VM ID

   需要改为只记录 request ID、阶段、耗时、计数和错误摘要。

5. **后端请求字段仍保留默认旧合同值**

6. **feature flag 目前只接了 Vite 环境变量**
